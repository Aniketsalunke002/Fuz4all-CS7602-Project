#include <map>

using namespace std;

class Employee {

public:
    Employee() = default;
    Employee(string name, int age) : name(name), age(age) {}
    string name;
    int age;
};

class Manager : public Employee {

public:
    Manager() = default;
    Manager(string name, int age, const map<string, Employee>& employees) : Employee(name, age), employees(employees) {}
    const map<string, Employee>& employees;
};

int main() {
    Manager manager("<NAME>", 30, {{"<NAME>", 18}, {"<NAME>", 30}});
    cout << manager.employees["<NAME>"] << endl;

    return 0;
}

#include <string>

class Person {
public:
    Person(std::string n) : name(n) {}
    Person(const Person &p) : name(p.name) {}
    Person &operator=(const Person &p) {
        name = p.name;
        return *this;
    }
    const std::string &getName() { return name; }
private:
    std::string name;
};

void printName(const Person &p) {
    std::cout << p.getName() << std::endl;
}

int main() {
    Person p("John");
    printName(p);
    p = "Jane";
    printName(p);
    return 0;
}
